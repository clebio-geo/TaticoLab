import sqlite3
import tkinter as tk
from tkinter import ttk, messagebox


DB_NAME = "gestao_3d.db"


class Database:
    def __init__(self, db_name):
        self.conn = sqlite3.connect(db_name)
        self.conn.row_factory = sqlite3.Row
        self.conn.execute("PRAGMA foreign_keys = ON")
        self.create_tables()
        self.update_schema()

    def create_tables(self):
        cursor = self.conn.cursor()

        cursor.execute("""
        CREATE TABLE IF NOT EXISTS impressoras (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nome TEXT NOT NULL,
            potencia_w REAL NOT NULL,
            custo_aquisicao REAL DEFAULT 0,
            vida_util_horas REAL DEFAULT 0,
            manutencao_mensal REAL DEFAULT 0
        )
        """)

        cursor.execute("""
        CREATE TABLE IF NOT EXISTS materiais (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nome TEXT NOT NULL,
            categoria TEXT,
            fabricante TEXT,
            tipo TEXT,
            cor TEXT,
            peso_rolo_g REAL DEFAULT 0,
            valor_rolo REAL DEFAULT 0,
            custo_unitario REAL DEFAULT 0,
            unidade TEXT
        )
        """)

        cursor.execute("""
        CREATE TABLE IF NOT EXISTS produtos (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nome TEXT NOT NULL,
            material_id INTEGER NOT NULL,
            impressora_id INTEGER NOT NULL,
            peso_peca_g REAL NOT NULL,
            tempo_impressao_h REAL NOT NULL,
            custo_insumos_extra REAL DEFAULT 0,
            custo_mao_obra REAL DEFAULT 0,
            percentual_falha REAL DEFAULT 0,
            margem_lucro REAL DEFAULT 0,
            preco_sugerido REAL DEFAULT 0,
            FOREIGN KEY(material_id) REFERENCES materiais(id),
            FOREIGN KEY(impressora_id) REFERENCES impressoras(id)
        )
        """)

        cursor.execute("""
        CREATE TABLE IF NOT EXISTS vendas (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            produto_id INTEGER NOT NULL,
            quantidade INTEGER NOT NULL,
            valor_unitario REAL NOT NULL,
            valor_total REAL NOT NULL,
            custo_unitario_produto REAL DEFAULT 0,
            custo_total_venda REAL DEFAULT 0,
            lucro_total REAL DEFAULT 0,
            data_venda TEXT DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY(produto_id) REFERENCES produtos(id)
        )
        """)

        self.conn.commit()

    def update_schema(self):
        colunas_materiais = {
            "categoria": "TEXT",
            "fabricante": "TEXT",
            "tipo": "TEXT",
            "cor": "TEXT",
            "peso_rolo_g": "REAL DEFAULT 0",
            "valor_rolo": "REAL DEFAULT 0",
            "custo_unitario": "REAL DEFAULT 0",
            "unidade": "TEXT"
        }

        colunas_vendas = {
            "custo_unitario_produto": "REAL DEFAULT 0",
            "custo_total_venda": "REAL DEFAULT 0",
            "lucro_total": "REAL DEFAULT 0"
        }

        existentes_materiais = self.execute("PRAGMA table_info(materiais)", fetch=True)
        nomes_materiais = [col["name"] for col in existentes_materiais]

        for coluna, tipo in colunas_materiais.items():
            if coluna not in nomes_materiais:
                self.execute(f"ALTER TABLE materiais ADD COLUMN {coluna} {tipo}")

        existentes_vendas = self.execute("PRAGMA table_info(vendas)", fetch=True)
        nomes_vendas = [col["name"] for col in existentes_vendas]

        for coluna, tipo in colunas_vendas.items():
            if coluna not in nomes_vendas:
                self.execute(f"ALTER TABLE vendas ADD COLUMN {coluna} {tipo}")

    def execute(self, query, params=(), fetch=False, fetchone=False):
        cursor = self.conn.cursor()
        cursor.execute(query, params)
        self.conn.commit()

        if fetchone:
            return cursor.fetchone()
        if fetch:
            return cursor.fetchall()
        return None

    def validar_texto_obrigatorio(self, valor, nome_campo):
        if not str(valor).strip():
            raise ValueError(f"Informe {nome_campo}.")

    def validar_numero_nao_negativo(self, valor, nome_campo):
        if valor < 0:
            raise ValueError(f"O campo '{nome_campo}' não pode ser negativo.")

    def validar_numero_positivo(self, valor, nome_campo):
        if valor <= 0:
            raise ValueError(f"O campo '{nome_campo}' deve ser maior que zero.")

    def alternar_campos_material(self, event=None):
        categoria = self.mat_categoria.get()

        if categoria == "Filamentos":
            self.frame_filamento.grid()
            self.lbl_peso_rolo.grid()
            self.entry_peso_rolo.grid()
            self.lbl_valor_rolo.grid()
            self.entry_valor_rolo.grid()
            self.lbl_custo_unitario.grid_remove()
            self.entry_custo_unitario.grid_remove()
            self.lbl_unidade.grid_remove()
            self.entry_unidade.grid_remove()
        else:
            self.frame_filamento.grid_remove()
            self.lbl_peso_rolo.grid_remove()
            self.entry_peso_rolo.grid_remove()
            self.lbl_valor_rolo.grid_remove()
            self.entry_valor_rolo.grid_remove()
            self.lbl_custo_unitario.grid(row=5, column=0, sticky="w", padx=5, pady=5)
            self.entry_custo_unitario.grid(row=5, column=1, padx=5, pady=5)
            self.lbl_unidade.grid(row=6, column=0, sticky="w", padx=5, pady=5)
            self.entry_unidade.grid(row=6, column=1, padx=5, pady=5)

    def custo_grama_material(self, material_id):
        mat = self.db.execute("SELECT * FROM materiais WHERE id = ?", (material_id,), fetchone=True)
        if not mat:
            return 0
        if mat["categoria"] == "Filamentos" and mat["peso_rolo_g"] > 0:
            return mat["valor_rolo"] / mat["peso_rolo_g"]
        return 0

    def custo_hora_impressora(self, impressora_id):
        imp = self.db.execute("SELECT * FROM impressoras WHERE id = ?", (impressora_id,), fetchone=True)
        if not imp:
            return 0

        energia_kwh_hora = imp["potencia_w"] / 1000
        custo_energia_hora = energia_kwh_hora * self.tarifa_energia.get()

        depreciacao_hora = 0
        if imp["vida_util_horas"] and imp["vida_util_horas"] > 0:
            depreciacao_hora = imp["custo_aquisicao"] / imp["vida_util_horas"]

        manutencao_hora = imp["manutencao_mensal"] / 160 if imp["manutencao_mensal"] else 0
        return custo_energia_hora + depreciacao_hora + manutencao_hora

    def calcular_preco(self, material_id, impressora_id, peso_g, tempo_h, insumos_extra, mao_obra, falha_pct, margem_pct):
        custo_material = self.custo_grama_material(material_id) * peso_g
        custo_impressao = self.custo_hora_impressora(impressora_id) * tempo_h
        subtotal = custo_material + custo_impressao + insumos_extra + mao_obra
        adicional_fixos = subtotal * (self.percentual_custos_fixos.get() / 100)
        subtotal += adicional_fixos
        fator_falha = 1 + (falha_pct / 100)
        custo_total = subtotal * fator_falha
        preco_sugerido = custo_total * (1 + margem_pct / 100)

        return {
            "custo_material": custo_material,
            "custo_impressao": custo_impressao,
            "custos_fixos": adicional_fixos,
            "custo_total": custo_total,
            "preco_sugerido": preco_sugerido
        }

    def registrar_venda(self):
        try:
            if not self.venda_produto.get():
                raise ValueError("Selecione um produto.")

            produto_id = int(self.venda_produto.get().split(" - ")[0])
            qtd = self.venda_quantidade.get()
            valor_unit = self.venda_valor_unitario.get()

            self.validar_numero_positivo(qtd, "Quantidade")
            self.validar_numero_positivo(valor_unit, "Valor unitário vendido")

            produto = self.db.execute("SELECT * FROM produtos WHERE id = ?", (produto_id,), fetchone=True)
            if not produto:
                raise ValueError("Produto não encontrado.")

            dados_custo = self.calcular_preco(
                material_id=produto["material_id"],
                impressora_id=produto["impressora_id"],
                peso_g=produto["peso_peca_g"],
                tempo_h=produto["tempo_impressao_h"],
                insumos_extra=produto["custo_insumos_extra"],
                mao_obra=produto["custo_mao_obra"],
                falha_pct=produto["percentual_falha"],
                margem_pct=produto["margem_lucro"]
            )

            custo_unitario = dados_custo["custo_total"]
            custo_total_venda = custo_unitario * qtd
            valor_total = qtd * valor_unit
            lucro_total = valor_total - custo_total_venda

            if self.venda_editando_id is None:
                self.db.execute("""
                    INSERT INTO vendas (
                        produto_id, quantidade, valor_unitario, valor_total,
                        custo_unitario_produto, custo_total_venda, lucro_total
                    )
                    VALUES (?, ?, ?, ?, ?, ?, ?)
                """, (produto_id, qtd, valor_unit, valor_total, custo_unitario, custo_total_venda, lucro_total))
                messagebox.showinfo("Sucesso", "Venda registrada com sucesso.")
            else:
                self.db.execute("""
                    UPDATE vendas
                    SET produto_id = ?, quantidade = ?, valor_unitario = ?, valor_total = ?,
                        custo_unitario_produto = ?, custo_total_venda = ?, lucro_total = ?
                    WHERE id = ?
                """, (produto_id, qtd, valor_unit, valor_total, custo_unitario, custo_total_venda, lucro_total, self.venda_editando_id))
                messagebox.showinfo("Sucesso", "Venda atualizada com sucesso.")

            self.cancelar_edicao_venda()
            self.atualizar_listas()
            self.gerar_relatorio()

    def gerar_relatorio(self):
        resumo = self.db.execute("""
            SELECT
                COUNT(*) AS qtd_vendas,
                COALESCE(SUM(valor_total), 0) AS faturamento_total,
                COALESCE(SUM(custo_total_venda), 0) AS custo_total,
                COALESCE(SUM(lucro_total), 0) AS lucro_total
            FROM vendas
        """, fetchone=True)

        produtos = self.db.execute("SELECT COUNT(*) AS qtd FROM produtos", fetchone=True)
        materiais = self.db.execute("SELECT COUNT(*) AS qtd FROM materiais", fetchone=True)
        impressoras = self.db.execute("SELECT COUNT(*) AS qtd FROM impressoras", fetchone=True)

        ranking_faturamento = self.db.execute("""
            SELECT
                p.nome,
                SUM(v.quantidade) AS qtd_total,
                SUM(v.valor_total) AS faturamento,
                SUM(v.lucro_total) AS lucro
            FROM vendas v
            JOIN produtos p ON p.id = v.produto_id
            GROUP BY p.id, p.nome
            ORDER BY faturamento DESC
            LIMIT 10
        """, fetch=True)

        ranking_lucro = self.db.execute("""
            SELECT
                p.nome,
                SUM(v.lucro_total) AS lucro,
                SUM(v.valor_total) AS faturamento
            FROM vendas v
            JOIN produtos p ON p.id = v.produto_id
            GROUP BY p.id, p.nome
            ORDER BY lucro DESC
            LIMIT 10
        """, fetch=True)

        produto_mais_lucrativo = self.db.execute("""
            SELECT
                p.nome,
                SUM(v.lucro_total) AS lucro
            FROM vendas v
            JOIN produtos p ON p.id = v.produto_id
            GROUP BY p.id, p.nome
            ORDER BY lucro DESC
            LIMIT 1
        """, fetchone=True)

        ticket_medio = 0
        if resumo["qtd_vendas"] > 0:
            ticket_medio = resumo["faturamento_total"] / resumo["qtd_vendas"]

        margem_real = 0
        if resumo["faturamento_total"] > 0:
            margem_real = (resumo["lucro_total"] / resumo["faturamento_total"]) * 100

        texto = []
        texto.append("RELATÓRIO GERENCIAL DO NEGÓCIO DE IMPRESSÃO 3D\n")
        texto.append(f"Faturamento total: R$ {resumo['faturamento_total']:.2f}")
        texto.append(f"Custo total estimado das vendas: R$ {resumo['custo_total']:.2f}")
        texto.append(f"Lucro estimado total: R$ {resumo['lucro_total']:.2f}")
        texto.append(f"Quantidade de vendas registradas: {resumo['qtd_vendas']}")
        texto.append(f"Ticket médio por venda: R$ {ticket_medio:.2f}")
        texto.append(f"Margem real estimada: {margem_real:.2f}%\n")

        texto.append(f"Produtos cadastrados: {produtos['qtd']}")
        texto.append(f"Materiais cadastrados: {materiais['qtd']}")
        texto.append(f"Impressoras cadastradas: {impressoras['qtd']}\n")

        if produto_mais_lucrativo:
            texto.append(f"Produto mais lucrativo: {produto_mais_lucrativo['nome']} | Lucro acumulado: R$ {produto_mais_lucrativo['lucro']:.2f}\n")
        else:
            texto.append("Produto mais lucrativo: ainda não há vendas registradas.\n")

        texto.append("RANKING DE PRODUTOS POR FATURAMENTO:")
        if ranking_faturamento:
            for i, item in enumerate(ranking_faturamento, start=1):
                texto.append(
                    f"{i}. {item['nome']} | Qtd: {item['qtd_total']} | "
                    f"Faturamento: R$ {item['faturamento']:.2f} | Lucro: R$ {(item['lucro'] or 0):.2f}"
                )
        else:
            texto.append("Nenhuma venda registrada ainda.")

        texto.append("\nRANKING DE PRODUTOS POR LUCRO:")
        if ranking_lucro:
            for i, item in enumerate(ranking_lucro, start=1):
                texto.append(
                    f"{i}. {item['nome']} | Lucro: R$ {(item['lucro'] or 0):.2f} | "
                    f"Faturamento: R$ {(item['faturamento'] or 0):.2f}"
                )
        else:
            texto.append("Nenhuma venda registrada ainda.")

        self.txt_relatorio.delete("1.0", tk.END)
        self.txt_relatorio.insert(tk.END, "\n".join(texto))

    def atualizar_listas(self):
        self.atualizar_tree_impressoras()
        self.atualizar_tree_materiais()
        self.atualizar_tree_produtos()
        self.atualizar_tree_vendas()
        self.atualizar_comboboxes()
        self.gerar_relatorio()

    def atualizar_tree_impressoras(self):
        for item in self.tree_impressoras.get_children():
            self.tree_impressoras.delete(item)

        dados = self.db.execute("SELECT * FROM impressoras ORDER BY id DESC", fetch=True)
        for row in dados:
            self.tree_impressoras.insert("", tk.END, values=(
                row["id"],
                row["nome"],
                row["potencia_w"],
                f"{row['custo_aquisicao']:.2f}",
                f"{row['vida_util_horas']:.2f}",
                f"{row['manutencao_mensal']:.2f}"
            ))

    def atualizar_tree_materiais(self):
        for item in self.tree_materiais.get_children():
            self.tree_materiais.delete(item)

        dados = self.db.execute("SELECT * FROM materiais ORDER BY id DESC", fetch=True)
        for row in dados:
            if row["categoria"] == "Filamentos" and row["peso_rolo_g"]:
                custo = row["valor_rolo"] / row["peso_rolo_g"]
                custo_texto = f"R$ {custo:.4f}/g"
            else:
                unidade = row["unidade"] if row["unidade"] else "un"
                custo_texto = f"R$ {row['custo_unitario']:.2f}/{unidade}"

            self.tree_materiais.insert("", tk.END, values=(
                row["id"],
                row["nome"],
                row["categoria"] or "-",
                row["fabricante"] or "-",
                row["tipo"] or "-",
                row["cor"] or "-",
                custo_texto
            ))

    def atualizar_tree_produtos(self):
        for item in self.tree_produtos.get_children():
            self.tree_produtos.delete(item)

        dados = self.db.execute("SELECT * FROM produtos ORDER BY id DESC", fetch=True)
        for row in dados:
            self.tree_produtos.insert("", tk.END, values=(
                row["id"],
                row["nome"],
                f"{row['preco_sugerido']:.2f}",
                row["peso_peca_g"],
                row["tempo_impressao_h"]
            ))

    def atualizar_tree_vendas(self):
        for item in self.tree_vendas.get_children():
            self.tree_vendas.delete(item)

        dados = self.db.execute("""
            SELECT v.id, p.nome AS produto, v.quantidade, v.valor_total,
                   v.custo_total_venda, v.lucro_total, v.data_venda
            FROM vendas v
            JOIN produtos p ON p.id = v.produto_id
            ORDER BY v.id DESC
        """, fetch=True)

        for row in dados:
            self.tree_vendas.insert("", tk.END, values=(
                row["id"],
                row["produto"],
                row["quantidade"],
                f"{(row['valor_total'] or 0):.2f}",
                f"{(row['custo_total_venda'] or 0):.2f}",
                f"{(row['lucro_total'] or 0):.2f}",
                row["data_venda"]
            ))

    def atualizar_comboboxes(self):
        materiais = self.db.execute("""
            SELECT id, nome, fabricante, tipo, cor
            FROM materiais
            WHERE categoria = 'Filamentos'
            ORDER BY nome
        """, fetch=True)
        impressoras = self.db.execute("SELECT id, nome FROM impressoras ORDER BY nome", fetch=True)
        produtos = self.db.execute("SELECT id, nome, preco_sugerido FROM produtos ORDER BY nome", fetch=True)

        lista_materiais = []
        for m in materiais:
            detalhes = " | ".join([x for x in [m["fabricante"], m["tipo"], m["cor"]] if x])
            if detalhes:
                lista_materiais.append(f"{m['id']} - {m['nome']} ({detalhes})")
            else:
                lista_materiais.append(f"{m['id']} - {m['nome']}")

        self.prod_material["values"] = lista_materiais
        self.prod_impressora["values"] = [f"{i['id']} - {i['nome']}" for i in impressoras]
        self.venda_produto["values"] = [f"{p['id']} - {p['nome']} (R$ {p['preco_sugerido']:.2f})" for p in produtos]

    def formatar_material_combobox_por_id(self, material_id):
        m = self.db.execute("SELECT id, nome, fabricante, tipo, cor FROM materiais WHERE id = ?", (material_id,), fetchone=True)
        if not m:
            return ""
        detalhes = " | ".join([x for x in [m["fabricante"], m["tipo"], m["cor"]] if x])
        return f"{m['id']} - {m['nome']} ({detalhes})" if detalhes else f"{m['id']} - {m['nome']}"

    def formatar_impressora_combobox_por_id(self, impressora_id):
        i = self.db.execute("SELECT id, nome FROM impressoras WHERE id = ?", (impressora_id,), fetchone=True)
        if not i:
            return ""
        return f"{i['id']} - {i['nome']}"

    def formatar_produto_combobox_por_id(self, produto_id):
        p = self.db.execute("SELECT id, nome, preco_sugerido FROM produtos WHERE id = ?", (produto_id,), fetchone=True)
        if not p:
            return ""
        return f"{p['id']} - {p['nome']} (R$ {p['preco_sugerido']:.2f})"
